  `anio_ejercicio` int not null,
  `periodo` int not null,
  `deptoId` int not null,
  `solicitado_por` int not null,
  `autorizado_por` int not null,
  `fecha_desde` datetime default null,
  `fecha_hasta` datetime default null,
  `razon_justificacion` varchar(500),
  `empresaid` int not null,
  `usuariocreated` varchar(20) collate utf8mb3_spanish_ci default null,
  `fechacreated` datetime default null,
  `usuarioupdated` varchar(20) collate utf8mb3_spanish_ci default null,
  `fechaupdated` datetime default null,
  primary key (`pleauthoraexmaeid`),
  key `fk_pleauthoraexmae_empresa` (`empresaid`),
  key `fk_create_pleauthoraexmae_tusuarios` (`usuariocreated`),
  key `fk_update_pleauthoraexmae_tusuarios` (`usuarioupdated`),
  constraint `fk_create_pleauthoraexmae_tusuarios` foreign key (`usuariocreated`) references `tusuarios` (`usuario`) on delete restrict on update cascade,
  constraint `fk_pleauthoraexmae_empresa` foreign key (`empresaid`) references `tempresas` (`empresaid`) on delete restrict on update cascade,
  constraint `fk_pleauthoraexmae_plempleados` foreign key (`solicitado_por`) references `plempleados` (`empleadoid`) on delete restrict on update cascade,

  constraint `fk_update_pleauthoraexmae_tusuarios` foreign key (`usuarioupdated`) references `tusuarios` (`usuario`) on delete restrict on update cascade
